    let fee, neto;
    if(isHTHT){
      // Para NatCash/MonCash Haití→Haití, el importe ingresado ya está en HTG.
      // La tarifa se consulta directamente en la tabla HTG, sin conversión DOP.
      fee = typeof window.v162_getFeeRetiroHT === 'function'
        ? Math.max(0, Number(window.v162_getFeeRetiroHT(monto)) || 0)
        : (monto >= 2000 && monto <= 3999 ? 25 : 0);
      neto = Math.max(0, monto - fee);
      if(netoEl) netoEl.textContent = neto.toFixed(2) + ' HTG';
      if(comEl) comEl.textContent = fee.toFixed(2) + ' HTG';
      if(recEl) recEl.textContent = neto.toFixed(2) + ' HTG';
      if(tasaEl) tasaEl.textContent = 'Tarifa Haití a Haití · Gross ' + monto.toFixed(2) + ' HTG · Comisión ' + fee.toFixed(2) + ' HTG · Neto ' + neto.toFixed(2) + ' HTG';
    } else {
      // Mantener la regla existente para las demás rutas; no convertir las rutas HT→HT.
      fee = Math.round(monto * 0.08);
      neto = Math.max(0, monto - fee);
      if(netoEl) netoEl.textContent = neto.toFixed(2) + ' RD$';
      if(comEl) comEl.textContent = fee.toFixed(2) + ' RD$';
      if(recEl) recEl.textContent = neto.toFixed(2) + ' RD$';
      if(tasaEl) tasaEl.textContent = 'Tasa 8% · ' + monto.toFixed(2) + ' RD$';
    }
  };

  // Sobrescribir funcion original
  window.v162_calcRetiro = window.v162_calcRetiro_CORREGIDO;
  window.v162_onRetiroRuta = window.v162_calcRetiro_CORREGIDO;
  window.v162_setRetiro = function(v){ 
    const el=document.getElementById('v162-retiro-ruta');
    if(el) el.value=v;
    window.v162_calcRetiro_CORREGIDO();
  };

  // Para envio tambien
  window.v162_calcEnvio_CORREGIDO = window.v162_calcRetiro_CORREGIDO;
  // Sobrescribir si existe
  if(window.calcularEnvio){
    const origCalcularEnvio = window.calcularEnvio;
    window.calcularEnvio = function(){
      const rutaEl = document.getElementById('envio-ruta');
      const ruta = rutaEl?.value||'';
      if(ruta.includes('HT → HT') || ruta.includes('Naca')){
        window.v162_calcRetiro_CORREGIDO();
        // Copiar valores a envio UI si existe
        const montoEl = document.getElementById('envio-monto');
        const netoEl = document.getElementById('envio-monto-neto');
        const comEl = document.getElementById('envio-comision');
        const recEl = document.getElementById('envio-a-recibir');
        if(montoEl && netoEl){
          const retNeto = document.getElementById('v162-retiro-neto')?.innerText||'';
          netoEl.innerText = retNeto;
        }
      } else {
        origCalcularEnvio();
      }
    };
  }

  // Inicializar
  setTimeout(()=>{ if(window.v162_calcRetiro_CORREGIDO) window.v162_calcRetiro_CORREGIDO(); }, 800);

  // Asegurar que deposito tiene su pagina individual con Excel PDF CSV buscador numeracion
  function asegurarDepositoPagina(){
    let viewDep = document.getElementById('view-depositos');
    if(!viewDep){
      // Buscar view-mibanco o cuentas y clonar estructura
      viewDep = document.getElementById('view-mibanco') || document.getElementById('view-cuentas');
    }
    // No tocamos estructura, solo habilitamos
    if(viewDep){
      viewDep.querySelectorAll('button, input, select').forEach(el=>{
        el.disabled=false;
        el.removeAttribute('disabled');
        el.style.pointerEvents='auto';
      });
    }
  }
  setTimeout(asegurarDepositoPagina, 1000);
})();
