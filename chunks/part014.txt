window.abrirModalSeparado = function(view){
  if(view==='envios') document.getElementById('modal-envio-compacto')?.classList.remove('hidden');
  else if(view==='retiros') {
    // Unificar todas las entradas de Agregar retiro con la ventanilla NatCash/MonCash
    // que contiene el cálculo correcto en gourdes haitianos (HTG).
    const modal=document.getElementById('modal-retiro-v162');
    if(modal){ modal.classList.remove('hidden'); if(typeof window.v162_onRetiroRuta==='function') window.v162_onRetiroRuta(); }
    else document.getElementById('modal-retiro-compacto')?.classList.remove('hidden');
  }
  else if(view==='depositos') document.getElementById('modal-deposito-exacto')?.classList.remove('hidden');
}
