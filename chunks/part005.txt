function addNotificacion(msg, tipo){
  tipo=tipo||'Envío';
  const noti={id:Date.now(), msg:msg, tipo:tipo, fecha:new Date().toLocaleString('es-DO'), leida:false};
  notificaciones.unshift(noti);
  document.getElementById('bell-count').innerText=notificaciones.filter(n=>!n.leida).length;
  document.getElementById('bell-count').classList.remove('hidden');
  const list=document.getElementById('notif-list');
  if(list){
    const div=document.createElement('div');
    div.className='p-3 border-b hover:bg-blue-50 cursor-pointer text-[11px]';
    div.innerHTML=`<div class="font-bold">🔔 ${msg}</div><div class="text-[9px]">Transacción: ${tipo} - ${noti.fecha} - Click lleva donde transacción - Solo campanita</div>`;
    div.onclick=function(){
      noti.leida=true;
      document.getElementById('notif-panel').classList.add('hidden');
      const routeType=String(tipo||'').trim().toLocaleLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,'');
      if(['recarga','recargas'].includes(routeType)) showView('recargas');
      else if(['paquetico','paqueticos'].includes(routeType)) showView('paquetikos');
      else if(['envio','envios'].includes(routeType)) showView('envios');
      else if(['retiro','retiros'].includes(routeType)) showView('retiros');
      else if(['deposito','depositos'].includes(routeType)) showView('depositos');
      else if(['factura','facturas','servicio','servicios'].includes(routeType)) showView('facturas');
      else if(['comision','comisiones'].includes(routeType)) showView('comisiones');
      else if(['ajuste','ajustes'].includes(routeType)) showView('ajustes');
      else if(['bono','bonos'].includes(routeType)) showView('bono');
      else if(['registro','remitente','remitentes'].includes(routeType)) showView('remitentes');
      else showView('tablero');
      notificaciones=notificaciones.filter(n=>n.id!==noti.id);
      document.getElementById('bell-count').innerText=notificaciones.filter(n=>!n.leida).length;
      div.remove();
    };
    list.prepend(div);
  }
}
