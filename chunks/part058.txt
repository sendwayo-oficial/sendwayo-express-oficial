  function exportar(view, formato){
    const tabla=tablaActual(view);
    if(!tabla){ alert('No hay tabla para exportar.'); return; }
    const nombre='Sendwayo-'+view+'-'+new Date().toISOString().slice(0,10);
    if(formato==='pdf'){
      const w=window.open('','_blank');
      if(!w){ alert('Permite las ventanas emergentes para guardar el PDF.'); return; }
      const html='<!doctype html><html><head><meta charset="utf-8"><title>'+nombre+'</title><style>body{font-family:Arial,sans-serif;padding:18px}h2{font-size:18px}table{border-collapse:collapse;width:100%;font-size:10px}th,td{border:1px solid #999;padding:5px;text-align:left}button{display:none}@media print{body{padding:0}}</style></head><body><h2>'+nombre+'</h2>'+tabla.outerHTML+'<script>window.onload=function(){window.print()}<\\/script></body></html>';
      w.document.open(); w.document.write(html); w.document.close(); return;
    }
    const rows=[...tabla.querySelectorAll('tr')].map(tr=>[...tr.querySelectorAll('th,td')].map(td=>'"'+td.innerText.replace(/\\s+/g,' ').trim().replace(/"/g,'""')+'"'));
    const csv='\\ufeff'+rows.map(r=>r.join(',')).join('\\r\\n');
    const blob=new Blob([csv],{type:formato==='excel'?'application/vnd.ms-excel;charset=utf-8':'text/csv;charset=utf-8'});
    const url=URL.createObjectURL(blob); const a=document.createElement('a'); a.href=url; a.download=nombre+(formato==='excel'?'.xls':'.csv'); document.body.appendChild(a); a.click(); a.remove(); setTimeout(()=>URL.revokeObjectURL(url),1000);
  }
  window.v162_exportarModulo=exportar;
  const renderOriginal=window.renderModuloSeparado;
  if(typeof renderOriginal==='function'){
    window.renderModuloSeparado=function(view){
      const result=renderOriginal.apply(this,arguments);
      if(['retiros','envios','depositos'].includes(view)){
        const cont=document.getElementById('view-'+view);
        if(cont)cont.querySelectorAll('button').forEach(btn=>{
          const t=(btn.textContent||'').trim().toLowerCase();
          if(t==='excel')btn.onclick=()=>exportar(view,'excel'); else if(t==='pdf')btn.onclick=()=>exportar(view,'pdf'); else if(t==='csv')btn.onclick=()=>exportar(view,'csv');
        });
      }
      return result;
    };
  }
})();
