// Función sugerencia al escribir número registrado
window.mostrarSugerencias = function(inputId, listaId){
  const input = document.getElementById(inputId);
  if(!input) return;
  const valor = input.value.toLowerCase();
  const lista = document.getElementById(listaId);
  if(!lista) return;
  
  if(valor.length < 2){
    lista.classList.add('hidden');
    return;
  }
  
  const filtrados = window.BENEFICIARIOS_DB.filter(b=> 
    b.nombre.toLowerCase().includes(valor) || 
    b.telefono.toLowerCase().includes(valor) ||
    b.id.toLowerCase().includes(valor)
  );
  
  if(filtrados.length===0){
    lista.classList.add('hidden');
    return;
  }
  
  lista.innerHTML='';
  filtrados.slice(0,5).forEach(b=>{
    const div=document.createElement('div');
    div.className='p-2 hover:bg-blue-50 cursor-pointer border-b text-[11px] flex justify-between';
    div.innerHTML=`<span class="font-bold">${b.nombre}</span><span class="text-gray-500">${b.telefono}</span>`;
    div.onclick=function(){
      input.value = b.nombre + ' - ' + b.telefono;
      // Si es campo nombre manual, también llenar tel manual
      if(inputId.includes('nombre-manual')){
        const telInput = document.getElementById(inputId.replace('nombre-manual','tel-manual'));
        if(telInput) telInput.value = b.telefono;
      }
      if(inputId.includes('tel-manual')){
        const nombreInput = document.getElementById(inputId.replace('tel-manual','nombre-manual'));
        if(nombreInput) nombreInput.value = b.nombre;
      }
      lista.classList.add('hidden');
    };
    lista.appendChild(div);
  });
  lista.classList.remove('hidden');
}
