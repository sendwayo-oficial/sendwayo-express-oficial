        ag.commissions.DOP+=m;

        ag.comisiones.DOP+=m;

          rem.commissions.DOP+=m;
          rem.commissions.HTG+=m;
          rem.comisiones.DOP+=m;
          rem.comisiones.HTG+=m;
        }
      }
    }
