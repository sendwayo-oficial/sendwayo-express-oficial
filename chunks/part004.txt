function abrirNuevo(tipo){
  const monto=prompt('Monto '+tipo+' DOP - Ej 100 - Si hago envío de una vez en envío debo ver eso pendiente:')||'100';
  if(!monto || isNaN(monto)) return;
  const montoNum=parseFloat(monto);
  if(currentUser.id==='ADMINADMIN') ADMIN_BAL.DOP-=montoNum;
  else if(currentUser.id.startsWith('AG')){ const ag=AGENTES.find(a=>a.id===currentUser.id); if(ag) ag.balances.DOP-=montoNum; ADMIN_BAL.DOP-=montoNum; }
  else{ const rem=REMITENTES.find(r=>r.id===currentUser.id); if(rem) rem.balances.DOP-=montoNum; }
  
  let montoRecibidoLimpio='';
  let fee=0;
  let neto=0;
  if(tipo==='Recarga'){
    // Recargas: siempre en pesos, sin quitarle nada - si son 100, pones 100 - sin comentario - queda interno
    montoRecibidoLimpio = montoNum + ' DOP';
    fee = 5;
    neto = montoNum;
  } else if(tipo==='Paqueticos' || tipo==='Paquetico' || tipo==='Paquetes'){
    montoRecibidoLimpio = montoNum + ' DOP';
    fee = 10;
    neto = montoNum;
  } else if(tipo==='Envío' || tipo.includes('RD') && tipo.includes('HT')){
    // RD a Haití: multiplicación por 2.1 - cantidad recibida en gourde haitiano
    fee = Math.round(montoNum*0.08);
