  function finishLogin() {
    const u = window.currentUser || (() => { try { return typeof currentUser !== 'undefined' ? currentUser : null; } catch (_) { return null; } })();
    if (u && (u.id || u.uid || u.codigo || u.email)) {
      successfulLoginThisPage = true;
      try { if (typeof currentUser !== 'undefined') currentUser = u; } catch (_) {}
      window.currentUser = u;
      document.body.classList.add('sendwayo-authenticated');
      document.body.classList.remove('sendwayo-login-open');
      const welcome = $('welcome-landing'); if (welcome) welcome.style.display = 'none';
      const login = $('login-screen'); if (login) { login.classList.add('hidden'); login.style.display = 'none'; }
      updateIdentity();
      installAccountSelectors();
      applyIdentityDefaults(true);
      hideFalseCredentialHints();
      refreshBellCount();
      // La primera pantalla privada siempre es el tablero, pero solo tras validar las credenciales.
      setTimeout(() => {
        if (document.body.classList.contains('sendwayo-authenticated') && typeof window.showView === 'function') window.showView('tablero');
        else { const dashboard = $('view-tablero'); if (dashboard) { document.querySelectorAll('.view').forEach(el => el.classList.add('hidden')); dashboard.classList.remove('hidden'); dashboard.style.display = 'block'; } }
      }, 0);
      return true;
    }
    document.body.classList.remove('sendwayo-authenticated');
    document.body.classList.add('sendwayo-login-open');
    const login = $('login-screen'); if (login) { login.classList.remove('hidden'); login.style.display = 'flex'; }
    const welcome = $('welcome-landing'); if (welcome) welcome.style.display = 'none';
    return false;
  }

  window.intentarLogin = function () {
    const username = txt($('login-usuario')?.value);
    const password = txt($('login-password')?.value);
    if (!username || !password) {
      document.body.classList.add('sendwayo-login-open');
      const err = $('login-error'); if (err) { err.textContent = 'Es obligatorio introducir usuario/correo e contraseña.'; err.classList.remove('hidden'); }
      if (!username) $('login-usuario')?.focus(); else $('login-password')?.focus();
      return;
    }
    // The existing credential validator remains the authority; the screen changes only if it sets a user.
    if (typeof priorLogin === 'function') priorLogin.apply(this, arguments);
    setTimeout(finishLogin, 120);
  };

    if ($('sendwayo-active-code')) $('sendwayo-active-code').textContent = code;

  }
