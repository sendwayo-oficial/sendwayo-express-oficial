// Agregar beneficiario manual con registro automático
window.agregarBeneficiarioManual = function(){
  const nombre = document.getElementById('envio-ben-nombre-manual')?.value||'';
  const tel = document.getElementById('envio-ben-tel-manual')?.value||'';
  if(!nombre && !tel){ alert('Escriba nombre manual o tel manual'); return; }
  
  const nuevo = {
    nombre: nombre||'Manual',
    telefono: tel||'000000',
    id:'BEN-'+Date.now().toString().slice(-4)
  };
  window.BENEFICIARIOS_DB.unshift(nuevo);
  
  // Actualizar select
  const sel=document.getElementById('envio-beneficiario');
  if(sel){
    const opt=document.createElement('option');
    opt.text = nuevo.nombre + ' - ' + nuevo.telefono;
    opt.selected=true;
    sel.prepend(opt);
  }
  
  alert('✅ Beneficiario agregado: '+nuevo.nombre+' - '+nuevo.telefono+' - Registro automático en sistema - Ahora aparece como sugerencia al escribir número');
}
