        if(opt && (!sel.value || !/HT\s*(?:→|->)\s*HT/i.test(sel.options[sel.selectedIndex]?.text||''))) sel.value=opt.value;
        if(typeof window.v162_onRetiroRuta==='function') window.v162_onRetiroRuta();

  window.sendwayoRetiroAction=function(action,id){
    const arr=transacciones();
    const t=arr.find(x=>String(x.id)===String(id));
    if(!t) return;
    if(action==='ver'){
      if(typeof window.ver==='function') window.ver(id); else alert('Retiro '+id+'\n'+(t.ruta||'')+'\n'+(t.monto||''));
      return;
    }
    if(action==='comprobante'){
      if(t.estado==='Aprobado' && typeof window.verComprobante==='function') window.verComprobante(id);
      else alert('El comprobante estará disponible después de aprobar el retiro.');
      return;
    }
    if(action==='aprobar'){
      if(typeof window.aprobar==='function') window.aprobar(id); else {t.estado='Aprobado';}
      guardarTransacciones();
      setTimeout(()=>{guardarTransacciones(); window.renderModuloSeparado('retiros');},80);
      return;
    }
    if(action==='rechazar'){
      t.estado='Rechazado'; t.actualizadoEn=new Date().toLocaleString('es-DO');
      guardarTransacciones(); if(typeof window.renderPrincipal==='function') window.renderPrincipal(); window.renderModuloSeparado('retiros'); return;
    }
    if(action==='noCoincide'){
      t.estado='No coincide'; t.actualizadoEn=new Date().toLocaleString('es-DO');
      guardarTransacciones(); if(typeof window.renderPrincipal==='function') window.renderPrincipal(); window.renderModuloSeparado('retiros'); return;
    }
    if(action==='modificar'){
      const actual=Number(t.montoValue)||0;
      const entrada=prompt('Modificar monto del retiro '+id+' ('+(t.esHT?'HTG':'DOP')+'). Monto actual: '+actual, String(actual));
      if(entrada===null || entrada.trim()==='') return;
      const monto=Number(entrada);
      if(!Number.isFinite(monto) || monto<=0){alert('Escribe un monto mayor que cero.');return;}
      const ht=!!t.esHT || /HT\s*→\s*HT|Naca a Naca|Montcash a Montcash|MonCash.*Tarifa por rango/i.test(normaliza(t.ruta)+' '+normaliza(t.rutaExacta));
      let fee=0;
      if(ht){
        if(typeof window.v162_getFeeRetiroHT==='function') fee=Number(window.v162_getFeeRetiroHT(monto))||0;
        else fee=monto<=1000?25:monto<=2000?50:monto<=3999?81:monto<=7999?130:monto<=11999?248:monto<=19999?342:monto<=39999?576:monto<=59999?945:monto<=74999?1260:monto<=99999?1440:Math.round(monto*.06);
      } else fee=Math.round(monto*.08);
      const neto=Math.max(0,monto-fee); const moneda=ht?'HTG':'DOP';
      t.montoValue=monto; t.monto=monto+' '+moneda; t.fee=fee; t.neto=neto; t.feeText='Comisión '+fee.toFixed(2)+' '+moneda; t.montoRecibido=neto.toFixed(2)+' '+moneda; t.moneda=ht?'GOURDES':'PESOS'; t.esHT=ht;
      if(t.estado==='No coincide' || t.estado==='Rechazado') t.estado='Pendiente';
      t.actualizadoEn=new Date().toLocaleString('es-DO'); guardarTransacciones(); if(typeof window.renderPrincipal==='function') window.renderPrincipal(); window.renderModuloSeparado('retiros'); return;
    }
    if(action==='borrar'){
      if(!confirm('¿Borrar el retiro '+id+'?')) return;
      const idx=arr.findIndex(x=>String(x.id)===String(id)); if(idx>=0) arr.splice(idx,1);
      guardarTransacciones(); if(typeof window.renderPrincipal==='function') window.renderPrincipal(); window.renderModuloSeparado('retiros');
    }
  };
