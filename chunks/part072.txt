        Object.assign(data,updates);

      await userRoot.child(key).update({authEmail:firebaseUser.email});
      data.authEmail=firebaseUser.email;

        if(actorRole==='REMITENTE'&&category==='accountWithdrawals')throw new Error('La regla actual de la base no permite reservar retiros de remitente. Debemos ampliar esa regla de forma segura antes de activar este módulo.');
        const key=safeDbKey(t.firebaseKey||id);const status=txt(t.estado||t.status||'Pendiente');
        const payload=transactionPayload(t,category,targetKey,status);
        updates[ROOT+'/'+category+'/'+key]=payload;
        created.push({t,category,key,payload,targetKey});
        if(actorRole==='REMITENTE'&&category!=='deposits')reservedRemitterTx.push({t,category,key,payload,targetKey});
      } else if(oldSig!==sig && t.firebaseCategory && t.firebaseKey){
        const category=String(t.firebaseCategory), key=safeDbKey(t.firebaseKey), status=txt(t.estado||t.status||'Pendiente');
        const patch={status,estado:status,updatedAtMs:Date.now()};
        if(['Aprobado','Rechazado','No coincide'].includes(status)){patch.reviewedByAuthUid=actorUid;patch.reviewedByUserId=state.profileKey;patch.reviewedByRole=actorRole;patch.reviewedAtMs=Date.now();}
        updates[ROOT+'/'+category+'/'+key+'/status']=patch.status;
        updates[ROOT+'/'+category+'/'+key+'/estado']=patch.estado;
        updates[ROOT+'/'+category+'/'+key+'/updatedAtMs']=patch.updatedAtMs;
        if(patch.reviewedByAuthUid){for(const [k,v] of Object.entries(patch))if(!['status','estado','updatedAtMs'].includes(k))updates[ROOT+'/'+category+'/'+key+'/'+k]=v;}
        txUpdates.push({t,category,key,patch});
      }
    }
    // Admin may adjust wallets. Agent wallet changes are limited to the shared admin wallet and must accompany an allowed pending/review transaction; Firebase rules must independently validate the same relationship.
    if(actorRole==='ADMIN'||actorRole==='AGENTE'){
      const agentWalletOperation=actorRole==='AGENTE'&&changes.current.some(t=>{
        const id=txt(t.id),old=state.lastTxSnapshot.get(id),status=txt(t.estado||t.status).toLowerCase(),category=categoryForTx(t);
        return ['envios','accountWithdrawals','recargas','paqueticos','pagos','paquetes'].includes(category)&&((old===undefined&&status==='pendiente')||(old!==undefined&&old!==transactionSignature(t)&&status==='aprobado'));
      });
      for(const account of trackedMoneyAccounts()){
        const key=txt(account.dbUserKey||account.rtdbUserKey||account.id);if(!key)continue;
        if(actorRole==='AGENTE'&&(key!=='admin'&&!['ADMINADMIN','ADMIN-000'].includes(txt(account.id))))continue;
        const signature=moneySignature(account);const old=state.lastAccountSnapshot.get(key);
        if(old!==undefined&&old!==signature){
          if(actorRole==='AGENTE'&&!agentWalletOperation)throw new Error('Un agente no puede modificar balances por separado. El cambio del balance compartido no está asociado a una operación autorizada.');
          const patch=profileMoneyPatch(account);
          if(actorRole==='AGENTE'){
            const related=changes.current.find(t=>{const id=txt(t.id),oldSig=state.lastTxSnapshot.get(id),status=txt(t.estado||t.status).toLowerCase();return ['envios','accountWithdrawals','recargas','paqueticos','pagos','paquetes'].includes(categoryForTx(t))&&((oldSig===undefined&&status==='pendiente')||(oldSig!==undefined&&oldSig!==transactionSignature(t)&&status==='aprobado'));});
            if(!related)throw new Error('No se pudo relacionar el balance compartido con la operación del agente. No se guardó.');
            patch.lastAgentFinancialActionId=txt(related.firebaseKey||related.id);patch.lastAgentFinancialActionCategory=txt(related.firebaseCategory||categoryForTx(related));patch.lastAgentFinancialActionAuthUid=actorUid;patch.lastAgentFinancialActionUserId=state.profileKey;patch.lastAgentFinancialActionAtMs=Date.now();
          }
          for(const [field,value] of Object.entries(patch))updates[ROOT+'/users/'+key+'/'+field]=value;
          accountChanges.push({key,account,patch,signature});
        }
      }
    }
    // A remittee's balance is decreased using the existing one-time reservation structure in Firebase rules.
    for(const item of reservedRemitterTx){
      const key=state.profileKey, profile=state.profile||{};
      const m=accountMoney(profile);const ui=localAccountByDbKey(key);const uiMoney=accountMoney(ui||{});const amount=transactionAmount(item.t);const currency=transactionCurrency(item.t);
      const prior=profile.balanceReservations&&profile.balanceReservations[item.key];
      if(prior&&prior.status==='RESERVED')throw new Error('Esta operación ya tiene una reserva de balance en Firebase; no se volverá a descontar.');
      const amountBefore=currency==='HTG'?(Number(profile.balanceCurrency==='HTG'?profile.balance:profile.balanceHTG)||0):Number(profile.balance??profile.balanceDOP??m.balances.DOP)||0;
      if(amountBefore<amount)throw new Error('El balance confirmado en Firebase no cubre esta operación. No se guardó ni se descontó dos veces.');
      const currDop=Number(profile.balance??profile.balanceDOP??m.balances.DOP)||0;
      const currHtg=Number(profile.balanceHTG??m.balances.HTG)||0;
      const nextDop=currency==='DOP'?Math.round((currDop-amount)*100)/100:currDop;
      const nextHtg=currency==='HTG'&&profile.balanceCurrency!=='HTG'?Math.round((currHtg-amount)*100)/100:currHtg;
      const nextBalance=currency==='HTG'&&profile.balanceCurrency==='HTG'?Math.round((currDop-amount)*100)/100:nextDop;
      if(nextBalance<0||nextHtg<0)throw new Error('El balance quedaría negativo. La operación no se guardó.');
      updates[ROOT+'/users/'+key+'/balance']=nextBalance;
      updates[ROOT+'/users/'+key+'/balanceHTG']=nextHtg;
      updates[ROOT+'/users/'+key+'/lastBalanceReservationId']=item.key;
      updates[ROOT+'/users/'+key+'/lastBalanceReservationCollection']=item.category;
      updates[ROOT+'/users/'+key+'/lastBalanceReservationCurrency']=currency;
      updates[ROOT+'/users/'+key+'/balanceReservations/'+item.key]={id:item.key,transactionId:item.key,authUid:actorUid,status:'RESERVED',collection:item.category,amount,currency,createdAtMs:Date.now()};
      updates[ROOT+'/users/'+key+'/balances/DOP']=uiMoney.balances.DOP;
      updates[ROOT+'/users/'+key+'/balances/HTG']=uiMoney.balances.HTG;
      item.payload.balanceReservationId=item.key;
      
      // Keep the new profile snapshot in memory; the transaction commits atomically with it below.
      accountChanges.push({key,account:ui||{},patch:{balance:nextBalance,balanceHTG:nextHtg},signature:moneySignature(ui||{})});
    }
    return {updates,created,txUpdates,accountChanges};
  }
  async function reconcileCloudChanges(){
    if(state.syncBusy||!state.readReady||!auth.currentUser||!state.profile)return;
    const current=getTransactions();const ids=new Set(current.map(t=>txt(t.id)));
    const changed=current.filter(t=>{const id=txt(t.id);return !!id&&(state.lastTxSnapshot.get(id)!==transactionSignature(t));});
    const deleted=[...state.lastTxSnapshot.keys()].filter(id=>!ids.has(id));
    const role=canonicalRole(state.profile,state.profileKey);
    const localMoneyChanged=trackedMoneyAccounts().some(a=>state.lastAccountSnapshot.get(txt(a.dbUserKey||a.rtdbUserKey||a.id))!==moneySignature(a));
    const accountChanged=(role==='ADMIN'||role==='AGENTE')&&localMoneyChanged;
    if(!changed.length&&!accountChanged&&!deleted.length){return;}
    state.syncBusy=true;
    try{
      const packageData=await buildCloudUpdates({current:changed,deleted});
      if(!Object.keys(packageData.updates).length){changed.forEach(t=>state.lastTxSnapshot.set(txt(t.id),transactionSignature(t)));return;}
      await db.ref().update(packageData.updates);
      for(const row of packageData.created){row.t.firebaseCategory=row.category;row.t.firebaseKey=row.key;row.t.dbUserKey=row.targetKey;row.t._cloudRaw=row.payload;state.lastTxSnapshot.set(txt(row.t.id),transactionSignature(row.t));if(!state.remoteTransactions[row.category])state.remoteTransactions[row.category]=[];state.remoteTransactions[row.category].push(row.t);}
      for(const row of packageData.txUpdates){row.t._cloudRaw={...(row.t._cloudRaw||{}),...row.patch};state.lastTxSnapshot.set(txt(row.t.id),transactionSignature(row.t));}
      for(const row of packageData.accountChanges){state.lastAccountSnapshot.set(row.key,row.signature);const stored=state.directory.find(r=>r.key===row.key);if(stored&&row.patch)Object.assign(stored.data,row.patch);}
      state.lastTxSnapshot=makeTransactionSnapshot();
      state.lastAccountSnapshot=makeAccountSnapshot();state.lastAccountValues=makeAccountValueSnapshot();
      if(state.directory.some(r=>r.key===state.profileKey)){try{const currentProfileSnap=await userRoot.child(state.profileKey).once('value');if(currentProfileSnap.exists())state.profile=currentProfileSnap.val();}catch(e){}}
      clearSyncIssue();
    }catch(error){
      const message=error?.code==='PERMISSION_DENIED'||/permission_denied|permission denied/i.test(error?.message||'')
        ? 'Firebase rechazó la escritura por sus reglas de seguridad. La operación no se confirmó; no la repita todavía.'
        : (error?.message||'No fue posible confirmar los cambios en Firebase.');
      showSyncIssue(message);
      // Revert the local transaction list to the last known remote baseline for any failed new record/status edit.
      try{restoreLocalAccounts();await loadCloudTransactions({key:state.profileKey,data:state.profile},canonicalRole(state.profile,state.profileKey),auth.currentUser.uid);state.lastTxSnapshot=makeTransactionSnapshot();}catch(e){}
    }finally{state.syncBusy=false;}
  }

  }
