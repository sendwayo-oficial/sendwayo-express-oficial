  window.guardarAjuste171 = function(){
    const cuenta=document.getElementById('aj171-cuenta').value;
    const moneda=document.getElementById('aj171-moneda').value;
    const tipo=document.getElementById('aj171-tipo').value;
    const cantidad=parseFloat(document.getElementById('aj171-cantidad').value)||0;
    const concepto=document.getElementById('aj171-concepto').value||'Ajuste';
    if(!cantidad){ alert('Pon cantidad de dinero'); return; }
    if(!cuenta){ alert('Selecciona remitente o admin'); return; }
    // Aplicar a balance
    let target = (window.REMITENTES||[]).find(r=>r.codigo===cuenta||r.id===cuenta);
    if(!target) target = (window.AGENTES||[]).find(a=>a.codigo===cuenta||a.id===cuenta);
    if(target){
      if(!target.balances) target.balances={DOP:0,HTG:0};
      if(!target.comisiones) target.comisiones={DOP:0,HTG:0};
      if(tipo==='BALANCE'){
        if(moneda==='PESOS') target.balances.DOP+=cantidad;
        else target.balances.HTG+=cantidad;
      } else {
        if(moneda==='PESOS') target.comisiones.DOP+=cantidad;
        else target.comisiones.HTG+=cantidad;
      }
    }
    // Mensaje
    alert('✅ '+tipo+' - '+cantidad+' '+(moneda==='PESOS'?'RD$ Pesos Dominicanos':'HTG Gourdes Haitianos')+' a '+cuenta+' - Concepto: '+concepto+' - Llega a cuenta - '+(moneda==='PESOS'?'Va a balance Pesos / Comisión RD$':'Va a balanza Gourdes / Comisión HTG'));
    document.getElementById('modal-ajuste-comision-v171').classList.add('hidden');
    // Notificacion
    if(window.addNotificacion) addNotificacion(tipo+' '+cantidad+' '+moneda+' a '+cuenta+' - '+concepto, 'Ajuste');
    // Actualizar tablero si existe
    if(window.renderPrincipal) renderPrincipal();
    if(window.actualizarTablero) actualizarTablero();
  };
